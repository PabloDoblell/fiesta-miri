<script>

  // Manejo del envío del formulario

  const form = document.getElementById('rsvpForm');

  const alertMsg = document.getElementById('alertMsg');


  form.addEventListener('submit', (e) => {

    e.preventDefault();


    const registro = {

      nombre: document.getElementById('nombre').value,

      personas: document.getElementById('personas').value,

      hora: document.getElementById('hora').value

    };


    // Obtener lista previa o inicializar vacía

    let registros = JSON.parse(localStorage.getItem('registrosMiri')) || [];

    registros.push(registro);


    // Guardar en localStorage

    localStorage.setItem('registrosMiri', JSON.stringify(registros));


    // Mostrar mensaje de éxito y limpiar campos

    alertMsg.style.display = 'block';

    form.reset();

    setTimeout(() => { alertMsg.style.display = 'none'; }, 4000);


    // Actualizar tabla si el panel admin está abierto

    cargarTablaAdmin();

  });


  // Lógica del Panel Administrador

  const toggleAdminBtn = document.getElementById('toggleAdminBtn');

  const adminPanel = document.getElementById('adminPanel');

  const clearDataBtn = document.getElementById('clearDataBtn');


  toggleAdminBtn.addEventListener('click', () => {

    if (adminPanel.style.display === 'none') {

      const password = prompt('Ingresa la contraseña de administrador:');

      // Contraseña predeterminada: 1234

      if (password === '1234') {

        adminPanel.style.display = 'block';

        cargarTablaAdmin();

      } else if (password !== null) {

        alert('Contraseña incorrecta');

      }

    } else {

      adminPanel.style.display = 'none';

    }

  });


  function cargarTablaAdmin() {

    const tbody = document.querySelector('#asistentesTable tbody');

    tbody.innerHTML = '';

    const registros = JSON.parse(localStorage.getItem('registrosMiri')) || [];


    if (registros.length === 0) {

      tbody.innerHTML = '<tr><td colspan="3" style="text-align:center;">Sin registros aún</td></tr>';

      return;

    }


    registros.forEach(reg => {

      const tr = document.createElement('tr');

      tr.innerHTML = `<td>${reg.nombre}</td><td>${reg.personas}</td><td>${reg.hora}</td>`;

      tbody.appendChild(tr);

    });

  }


  clearDataBtn.addEventListener('click', () => {

    if (confirm('¿Estás seguro de que deseas borrar todos los registros?')) {

      localStorage.removeItem('registrosMiri');

      cargarTablaAdmin();

    }

  });

</script>
